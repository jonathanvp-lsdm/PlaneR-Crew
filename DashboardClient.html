  • ✅ Aucune alerte.
    • `; return; } alertes.forEach(function(alerte){ let icone = "ℹ️"; switch(alerte.niveau){ case "critical": icone = "🔴"; break; case "warning": icone = "🟠"; break; case "success": icone = "🟢"; break; } container.innerHTML += `
  • ${icone} ${alerte.texte}
    • `; }); }
